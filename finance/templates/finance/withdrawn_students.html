@login_required @user_passes_test(lambda u: u.is_superuser) def withdraw_student(request, student_id): student = get_object_or_404(Student, id=student_id) current_year = student.academic_year total_paid = Payment.objects.filter( student=student, academic_year=current_year, is_cancelled=False ).aggregate(s=Sum('amount_paid'))['s'] or Decimal('0.00') # التقاط الشاشة المرجعية أثناء العرض (GET) referer = request.META.get('HTTP_REFERER', '') from_screen = request.GET.get('from') or request.GET.get('next') if not from_screen: if 'quick-collection' in referer or 'quick_collection' in referer: from_screen = 'quick_collection' elif 'counter' in referer: from_screen = 'counter' if request.method == "POST": refund_amount = Decimal(request.POST.get('refund_amount', '0')) reason = request.POST.get('reason', 'سحب ملف نهائي') from_screen_post = request.POST.get('from_screen') or from_screen try: process_student_withdrawal( student_id=student.id, refund_amount=refund_amount, admin_user=request.user, reason=reason ) messages.success(request, f"✅ تم سحب ملف الطالب وتصفير مديونيته بنجاح. المبلغ المسترد: {refund_amount} ج.م") cache.clear() # التوجيه الذكي بناءً على الشاشة السابقة if from_screen_post == 'quick_collection' or 'quick-collection' in referer or 'quick_collection' in referer: try: target_base = reverse('quick_collection') except Exception: target_base = '/finance/quick-collection/' elif 'counter' in referer or from_screen_post == 'counter': try: target_base = reverse('counter') except Exception: target_base = '/finance/counter/' else: target_base = reverse('student_list') if 'student_list' not in target_base: target_url = f"{target_base}?student_id={student.id}" else: target_url = target_base return redirect(target_url) except Exception as e: messages.error(request, f"❌ حدث خطأ: {e}") return render(request, 'students/withdraw_student.html', { 'student': student, 'total_paid': total_paid, 'from_screen': from_screen })
